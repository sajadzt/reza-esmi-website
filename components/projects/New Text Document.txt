"use client";

import Image from "next/image";
import { useEffect, useMemo, useRef, useState } from "react";

interface ProjectVisualProps {
  hero: string;
  gallery?: string[];
  plans?: string[];
  title?: string;
}

export default function ProjectVisual({
  hero,
  gallery = [],
  plans = [],
  title = "Project",
}: ProjectVisualProps) {
  const images = useMemo(() => {
    return Array.from(
      new Set(
        [...gallery, ...plans].filter(
          (src): src is string =>
            typeof src === "string" && src.trim().length > 0
        )
      )
    );
  }, [gallery, plans]);

  const [selectedImage, setSelectedImage] = useState(hero);
  const [selectedIndex, setSelectedIndex] = useState(-1);

  const galleryRef = useRef<HTMLDivElement>(null);

  /*
   * Keep hero synchronized if the project changes.
   */
  useEffect(() => {
    setSelectedImage(hero);
    setSelectedIndex(-1);
  }, [hero]);

  /*
   * Horizontal mouse wheel.
   *
   * The page itself will still scroll normally.
   * Only the gallery converts wheel movement to horizontal movement.
   */
  useEffect(() => {
    const element = galleryRef.current;

    if (!element) return;

    const handleWheel = (event: WheelEvent) => {
      const horizontal =
        Math.abs(event.deltaX) > Math.abs(event.deltaY);

      if (!horizontal) {
        event.preventDefault();
        element.scrollLeft += event.deltaY;
      }
    };

    element.addEventListener("wheel", handleWheel, {
      passive: false,
    });

    return () => {
      element.removeEventListener("wheel", handleWheel);
    };
  }, []);

  const handleSelect = (src: string, index: number) => {
    setSelectedImage(src);
    setSelectedIndex(index);
  };

  return (
    <section className="relative w-full">
      {/* =====================================================
          MAIN PROJECT IMAGE
      ===================================================== */}

      <div className="mx-auto w-full max-w-[1500px] px-4 md:px-8">
        <div className="relative aspect-[16/9] w-full overflow-hidden bg-black">
          <Image
            key={selectedImage}
            src={selectedImage}
            alt={`${title} selected image`}
            fill
            priority
            sizes="(max-width: 768px) 100vw, 1500px"
            className="object-contain transition-all duration-500 ease-out"
          />

          {/* cinematic overlay */}
          <div className="pointer-events-none absolute inset-0 bg-gradient-to-t from-black/45 via-transparent to-black/10" />

          {/* project counter */}
          <div className="absolute bottom-5 left-5 z-10 flex items-center gap-3">
            <span className="h-1.5 w-1.5 rounded-full bg-[#bc3019]" />

            <span className="text-[9px] uppercase tracking-[0.3em] text-white/60">
              {selectedIndex >= 0
                ? `${String(selectedIndex + 1).padStart(2, "0")} / ${String(
                    images.length
                  ).padStart(2, "0")}`
                : "Project"}
            </span>
          </div>
        </div>
      </div>

      {/* =====================================================
          HORIZONTAL GALLERY
      ===================================================== */}

      {images.length > 0 && (
        <div className="mt-8 w-full">
          <div
            ref={galleryRef}
            className="
              flex
              w-full
              flex-nowrap
              gap-5
              overflow-x-auto
              overflow-y-hidden
              px-[8vw]
              py-5
              snap-x
              snap-mandatory
              overscroll-x-contain
              [scrollbar-width:none]
            "
            style={{
              WebkitOverflowScrolling: "touch",
            }}
          >
            {images.map((src, index) => {
              const selected = selectedIndex === index;

              return (
                <button
                  key={`${src}-${index}`}
                  type="button"
                  onClick={() => handleSelect(src, index)}
                  className={`
                    relative
                    flex-none
                    snap-center
                    overflow-visible
                    text-left
                    transition-all
                    duration-500
                    ease-out
                    ${
                      selected
                        ? "w-[min(46vw,600px)] scale-[1.04] opacity-100"
                        : "w-[min(28vw,360px)] scale-100 opacity-40 hover:opacity-80"
                    }
                  `}
                  aria-label={`Select project image ${index + 1}`}
                >
                  <div className="relative aspect-[4/3] w-full overflow-hidden bg-white/[0.025]">
                    <Image
                      src={src}
                      alt={`${title} image ${index + 1}`}
                      fill
                      sizes="(max-width: 768px) 70vw, 600px"
                      className={`
                        object-contain
                        transition-all
                        duration-500
                        ${
                          selected
                            ? "brightness-100"
                            : "brightness-75"
                        }
                      `}
                    />

                    {/* selected backlight */}
                    <div
                      className={`
                        pointer-events-none
                        absolute
                        -inset-3
                        rounded-sm
                        transition-all
                        duration-500
                        ${
                          selected
                            ? "bg-white/[0.045] opacity-100 blur-xl"
                            : "opacity-0"
                        }
                      `}
                    />

                    {/* selected border */}
                    <div
                      className={`
                        pointer-events-none
                        absolute
                        inset-0
                        border
                        transition-all
                        duration-500
                        ${
                          selected
                            ? "border-white/60"
                            : "border-white/10"
                        }
                      `}
                    />

                    {/* image number */}
                    <span
                      className={`
                        absolute
                        bottom-3
                        left-3
                        z-10
                        text-[9px]
                        uppercase
                        tracking-[0.3em]
                        ${
                          selected
                            ? "text-white"
                            : "text-white/40"
                        }
                      `}
                    >
                      {String(index + 1).padStart(2, "0")}
                    </span>
                  </div>
                </button>
              );
            })}
          </div>

          {/* gallery information */}
          <div className="flex items-center justify-between px-[8vw] text-[8px] uppercase tracking-[0.3em] text-white/25">
            <span>Project documentation</span>

            <span>Scroll / Drag / Select</span>
          </div>
        </div>
      )}
    </section>
  );
}
